# 3D View (Three.js Toggle) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A toggleable 3D battlefield view alongside the existing 2D Pixi canvas. Both renderers receive the same telemetry; only one is visible at a time. Drones have an altitude dimension (z), beams are 3D lines, turrets rotate 3D models.

**Architecture:** `Battlefield3D` mirrors the `Battlefield` interface (`init()`, `pushFrame()`, `destroy()`). `RunController` gets an optional `battlefield3d` reference. A "2D/3D" toggle button in the topbar swaps `display:none` on the two canvas containers. The backend adds optional `z` to `DroneFrame` (backward-compatible). Drone z linearly interpolates from scenario altitude to 0 as TTI decreases.

**Tech Stack:** Three.js r165+, `@types/three`, `three/addons/controls/OrbitControls.js`, `three/addons/lines/Line2.js`

- [ ] **Step 5: Compute z in telemetry builder**

Find where `DroneFrame` objects are created in `backend/beam/engine/telemetry.py`. The z should linearly interpolate from `drone_altitude` (when drone spawns, TTI is longest) to 0 (at asset). Since we track current TTI, approximate:

```python
# In the function that builds DroneFrame objects — locate the drone frame construction
# and add z computation. After determining tti for the drone:
#
#   z = cfg.sim.drone_altitude * (tti / initial_tti)  # needs initial TTI
#
# Simpler: z = cfg.sim.drone_altitude * hp_frac  (proxy for time alive)
# Even simpler: z = cfg.sim.drone_altitude * max(0, min(1, tti / max_tti_cap))
#
# Implementation: use tti / TTI_CAP where TTI_CAP = spawn_radius / speed as proxy.
# For the plan: use (tti / 60.0) clamped to [0,1] * drone_altitude
# (60s is a conservative TTI cap; works for typical scenario speeds).

z = float(cfg.sim.drone_altitude * min(1.0, max(0.0, drone.tti / 60.0)))
```

Then pass `z=z` when constructing `DroneFrame`.

Note: `telemetry.py` may not have access to `cfg` directly. Check how it is called and thread the config through if needed. If `telemetry.py` uses a separate `build_frame()` function, add a `drone_altitude: float = 0.0` parameter to that function.

```typescript
// frontend/src/render/battlefield3d.ts
import * as THREE from "three";
import type { FrameMessage } from "../types";

// Lazy-load OrbitControls to avoid import issues in test environment
let OrbitControls: new (camera: THREE.Camera, element: HTMLElement) => { update(): void; dispose(): void };

export interface Battlefield3DOptions {
  parent: HTMLElement;
}

export class Battlefield3D {
  private readonly parent: HTMLElement;
  private renderer: THREE.WebGLRenderer | null = null;
  private scene: THREE.Scene | null = null;
  private camera: THREE.PerspectiveCamera | null = null;
  private controls: { update(): void; dispose(): void } | null = null;
  private rafId: number | null = null;
  private active = false;

  // Entity pools
  private dronePool: Map<string, THREE.Mesh> = new Map();
  private turretPool: Map<string, { base: THREE.Mesh; barrel: THREE.Mesh }> = new Map();
  private beamPool: Map<string, THREE.Line> = new Map();

  // Shared geometries (created once)
  private droneGeo!: THREE.ConeGeometry;
  private turretBaseGeo!: THREE.BoxGeometry;
  private turretBarrelGeo!: THREE.CylinderGeometry;
  private assetMesh!: THREE.Mesh;

  private readonly SCALE = 0.1; // world units per metre

  constructor(opts: Battlefield3DOptions) {
    this.parent = opts.parent;
  }

  async init(): Promise<void> {
    if (this.renderer) return;

    this.renderer = new THREE.WebGLRenderer({ antialias: true });
    this.renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
    this.renderer.setSize(this.parent.clientWidth || 800, this.parent.clientHeight || 600);
    this.renderer.domElement.style.width = "100%";
    this.renderer.domElement.style.height = "100%";
    this.parent.appendChild(this.renderer.domElement);

    this.scene = new THREE.Scene();
    this.scene.background = new THREE.Color(0x0a0e14);

    this.camera = new THREE.PerspectiveCamera(
      50,
      (this.parent.clientWidth || 800) / (this.parent.clientHeight || 600),
      1,
      50000,
    );
    this.camera.position.set(0, 800, 600);
    this.camera.lookAt(0, 0, 0);

    // Lighting
    this.scene.add(new THREE.AmbientLight(0x334455, 2));
    const sun = new THREE.DirectionalLight(0xffffff, 1.5);
    sun.position.set(200, 400, 100);
    this.scene.add(sun);

    // Ground grid
    const grid = new THREE.GridHelper(2000, 40, 0x1a2030, 0x1a2030);
    this.scene.add(grid);

    // Asset (defended point)
    const assetGeo = new THREE.CylinderGeometry(8, 8, 20, 16);
    const assetMat = new THREE.MeshStandardMaterial({ color: 0x00ff88, emissive: 0x003311 });
    this.assetMesh = new THREE.Mesh(assetGeo, assetMat);
    this.scene.add(this.assetMesh);

    // Shared geometries for pools
    this.droneGeo = new THREE.ConeGeometry(3, 8, 6);
    this.turretBaseGeo = new THREE.BoxGeometry(10, 6, 10);
    this.turretBarrelGeo = new THREE.CylinderGeometry(1.5, 1.5, 20, 8);

    // Lazy load OrbitControls
    try {
      const mod = await import("three/addons/controls/OrbitControls.js");
      OrbitControls = mod.OrbitControls;
      this.controls = new OrbitControls(this.camera, this.renderer.domElement);
    } catch {
      // OrbitControls not available in test env — skip
    }
  }

  setActive(active: boolean): void {
    this.active = active;
    if (active) {
      this.startLoop();
    } else {
      this.stopLoop();
    }
  }

  pushFrame(msg: FrameMessage): void {
    if (!this.scene) return;
    this._syncDrones(msg);
    this._syncTurrets(msg);
    this._syncBeams(msg);
  }

  private _syncDrones(msg: FrameMessage): void {
    const seen = new Set<string>();
    for (const d of msg.drones) {
      seen.add(d.id);
      let mesh = this.dronePool.get(d.id);
      if (!mesh) {
        const mat = new THREE.MeshStandardMaterial({
          color: d.value > 5000 ? 0xffaa00 : 0xcccccc,
        });
        mesh = new THREE.Mesh(this.droneGeo, mat);
        this.scene!.add(mesh);
        this.dronePool.set(d.id, mesh);
      }
      mesh.position.set(d.x * this.SCALE, (d.z ?? 0) * this.SCALE, d.y * this.SCALE);
      mesh.visible = d.state !== "dead" && d.state !== "leaked";
      (mesh.material as THREE.MeshStandardMaterial).opacity = d.hp_frac;
    }
    for (const [id, mesh] of this.dronePool) {
      if (!seen.has(id)) {
        this.scene!.remove(mesh);
        this.dronePool.delete(id);
      }
    }
  }

  private _syncTurrets(msg: FrameMessage): void {
    const seen = new Set<string>();
    for (const t of msg.turrets) {
      seen.add(t.id);
      let group = this.turretPool.get(t.id);
      if (!group) {
        const baseMat = new THREE.MeshStandardMaterial({ color: 0x334455 });
        const barrelMat = new THREE.MeshStandardMaterial({ color: 0x00ff88 });
        const base = new THREE.Mesh(this.turretBaseGeo, baseMat);
        const barrel = new THREE.Mesh(this.turretBarrelGeo, barrelMat);
        barrel.rotation.z = Math.PI / 2;
        barrel.position.x = 12;
        this.scene!.add(base);
        this.scene!.add(barrel);
        group = { base, barrel };
        this.turretPool.set(t.id, group);
      }
      group.base.position.set(t.x * this.SCALE, 3, t.y * this.SCALE);
      group.barrel.position.set(
        t.x * this.SCALE + Math.cos(t.aim) * 12,
        3,
        t.y * this.SCALE + Math.sin(t.aim) * 12,
      );
      group.barrel.rotation.y = -t.aim;
    }
    for (const [id, group] of this.turretPool) {
      if (!seen.has(id)) {
        this.scene!.remove(group.base);
        this.scene!.remove(group.barrel);
        this.turretPool.delete(id);
      }
    }
  }

  private _syncBeams(msg: FrameMessage): void {
    const seen = new Set<string>();
    for (const b of msg.beams) {
      const key = `${b.fromTurret}-${b.to}`;
      seen.add(key);

      const turret = msg.turrets.find((t) => t.id === b.fromTurret);
      const drone = msg.drones.find((d) => d.id === b.to);
      if (!turret || !drone) continue;

      let line = this.beamPool.get(key);
      if (!line) {
        const geo = new THREE.BufferGeometry();
        const mat = new THREE.LineBasicMaterial({ color: 0x00ff88 });
        line = new THREE.Line(geo, mat);
        this.scene!.add(line);
        this.beamPool.set(key, line);
      }
      const points = [
        new THREE.Vector3(turret.x * this.SCALE, 3, turret.y * this.SCALE),
        new THREE.Vector3(drone.x * this.SCALE, (drone.z ?? 0) * this.SCALE, drone.y * this.SCALE),
      ];
      (line.geometry as THREE.BufferGeometry).setFromPoints(points);
      (line.material as THREE.LineBasicMaterial).opacity = b.power_frac;
    }
    for (const [key, line] of this.beamPool) {
      if (!seen.has(key)) {
        this.scene!.remove(line);
        this.beamPool.delete(key);
      }
    }
  }

  private startLoop(): void {
    if (this.rafId !== null) return;
    const tick = () => {
      this.controls?.update();
      if (this.renderer && this.scene && this.camera) {
        this.renderer.render(this.scene, this.camera);
      }
      this.rafId = requestAnimationFrame(tick);
    };
    this.rafId = requestAnimationFrame(tick);
  }

  private stopLoop(): void {
    if (this.rafId !== null) cancelAnimationFrame(this.rafId);
    this.rafId = null;
  }

  destroy(): void {
    this.stopLoop();
    this.controls?.dispose();
    this.renderer?.dispose();
    if (this.renderer && this.parent.contains(this.renderer.domElement)) {
      this.parent.removeChild(this.renderer.domElement);
    }
    this.renderer = null;
    this.scene = null;
    this.camera = null;
  }
}
```
